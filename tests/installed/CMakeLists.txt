cmake_minimum_required(VERSION 3.22)
project(evpl_installed_consumer C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_path(EVPL_INCLUDE_DIR evpl/evpl.h REQUIRED)
find_library(EVPL_LIBRARY evpl REQUIRED)
find_library(EVPL_HTTP_LIBRARY evpl_http REQUIRED)
add_executable(consumer main.c)
target_include_directories(consumer PRIVATE ${EVPL_INCLUDE_DIR})
target_link_libraries(consumer PRIVATE ${EVPL_LIBRARY} ${EVPL_HTTP_LIBRARY})
if(MSVC)
    target_compile_options(consumer PRIVATE /experimental:c11atomics /W3 /WX)
endif()
